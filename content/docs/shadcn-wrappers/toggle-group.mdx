## Usage

### Basic
```tsx
import { ToggleGroupWrapper } from "@/components/ui/toggle-group";

export function Basic() {
  return (
     <ToggleGroupWrapper
      options={["Toogle 1", "Toggle 2"]}
    />
  )
}
```

  /**
   * Array of toggle options to render
   * 
   * Supports:
   * - Primitives: \`string | number | boolean\`

   * \`\`\`tsx

   *   { label: <ItalicIcon />, value: "italic", className: "text-blue-500" }
   * ]
   * \`\`\`
   */
  options: toggleItemsT;

  /**
   * Additional CSS classes applied to all toggle items
   * Individual options can override this with their own \`className\`
   *

  /**
   * Controls toggle behavior

   * \`\`\`tsx
   * type="multiple" // Allow multiple selections
   * \`\`\`
   */
  type?: "single" | "multiple";

  /**
   * ...otherProps: All properties from Base UI ToggleGroup Root can be passed directly.
