## Usage

### Basic

```tsx
import { DropdownWrapper } from "@/components/ui/dropdown-menu-wrapper"
import { buttonVariants } from "@/components/ui/button"

<DropdownWrapper
  options={["Edit", "Duplicate", "---", "Delete"]}
  trigger="Actions"
  triggerCls={buttonVariants({ variant: "outline" })}
/>
```

### Shortcuts and destructive variant

```tsx
<DropdownWrapper
  options={[
    { label: "Edit", value: "edit", shortcut: "⌘E" },
    { label: "Delete", value: "delete", variant: "destructive", shortcut: "⌘⌫" },
  ]}
  trigger="Actions"
  triggerCls={buttonVariants({ variant: "outline" })}
  onSelect={v => console.log(v)}
/>
```

### Checkbox

```tsx
const [checked, setChecked] = useState<allowedPrimitiveT[]>([])

<DropdownCheckboxWrapper
  options={["Auto-save", "Spellcheck", "Dark mode"]}
  checked={checked}
  onCheckedChange={(v, c) =>
    setChecked(prev => (c ? [...prev, v] : prev.filter(x => x !== v)))
  }
  trigger="Features"
  triggerCls={buttonVariants({ variant: "outline" })}
/>
```

### Radio

```tsx
const [val, setVal] = useState<allowedPrimitiveT>("Light")

<DropdownRadioWrapper
  options={["Light", "Dark", "System"]}
  value={val}
  onValueChange={setVal}
  trigger={`Theme: ${val}`}
  triggerCls={buttonVariants({ variant: "outline" })}
/>
```

### Indicator position

```tsx
<DropdownCheckboxWrapper options={options} indicatorAt="right" trigger="..." />
<DropdownRadioWrapper options={options} indicatorAt="right" trigger="..." />
```

### Groups, submenus, separators

```tsx
const options: menuOptionsT = [
  { label: "New File", value: "new", shortcut: "⌘N" },
  "Save",
  "---",
  {
    group: "Settings",
    options: [{ label: "Appearance", value: "appearance" }, "Shortcuts"],
  },
  {
    submenu: "Export",
    options: ["PDF", "HTML", { group: "Advanced", options: ["LaTeX"] }],
  },
]
```

name="DropdownWrapper"
type={`export interface DropdownWrapper {
  /**
   * Trigger element content — rendered inside a button.
   */
  trigger: React.ReactNode;

  /**
   * CSS class applied to the trigger button.
   */
  triggerCls?: string;

  /**
   * Props forwarded to DropdownMenuTrigger — use for render prop (custom element), disabled, etc.
   *
   * \`\`\`tsx
   * triggerProps={{ render: <div role="button" tabIndex={0} /> }}
   * \`\`\`
   */
  triggerProps?: Omit<React.ComponentProps<typeof DropdownMenuTrigger>, "children" | "className">;

  /**
   * Array of menu options. Supports primitives, separator "---", objects, groups, and submenus.
   *
   * \`\`\`tsx
   * options={[
   *   "Simple item",
   *   { label: "Edit", value: "edit", shortcut: "⌘E" },
   *   { label: "Delete", value: "delete", variant: "destructive" },
   *   "---",
   *   { group: "Actions", options: ["Save", "Export"] },
   *   { submenu: "Share", options: ["Email", "Link"] }
   * ]}
   * \`\`\`
   */
  options: menuOptionsT;

  /**
   * CSS class applied to all menu items.
   */
  itemCls?: string;

  /**
   * CSS class applied to all groups.
   */
  groupCls?: string;

  /**
   * CSS class applied to all group labels.
   */
  groupLabelCls?: string;

  /**
   * Props forwarded to DropdownMenuContent.
   *
   * \`\`\`tsx
   * contentProps={{ align: "end", className: "w-56" }}
   * \`\`\`
   */
  contentProps?: React.ComponentProps<typeof DropdownMenuContent>;

  /**
   * Callback fired when an item is selected.
   */
  onSelect?: (value: allowedPrimitiveT) => void;

  /**
   * ...rootProps: All DropdownMenu Root props forwarded to the root.
   */
  otherProps?: never;
}`}
/>
