## Usage

### Basic

```tsx
import { DialogWrapper } from "@/components/ui/dialog"

export function Basic() {
  return (
    <DialogWrapper
      trigger="Open"
      triggerCls={buttonVariants({ variant: 'outline' })}
      title="Edit profile"
      description="Make changes to your profile here."
      action="Save"
    />
  )
}
```

### Icon trigger

```tsx
<DialogWrapper
  trigger={<><Settings /> Settings</>}
  triggerCls={buttonVariants({ variant: 'secondary' })}
  title="Settings"
  action="Save changes"
/>
```

### Custom element trigger

Use `triggerProps.render` to replace the default `<button>` with any element:

```tsx
<DialogWrapper
  trigger="Open"
  triggerProps={{
    render: <a href="#" />,
    nativeButton: false,
  }}
  triggerCls="text-sm text-primary underline underline-offset-4"
  title="Opened via anchor"
/>
```

### Hide close button

```tsx
<DialogWrapper
  trigger="Open"
  title="No close button"
  showCloseButton={false}
  action="Done"
/>
```

### Controlled

```tsx
export function Controlled() {
  const [open, setOpen] = useState(false)

  return (
    <>
      <button onClick={() => setOpen(true)}>Open</button>

      <DialogWrapper
        open={open}
        onOpenChange={setOpen}
        title="Controlled dialog"
        action="Confirm"
        onAction={() => setOpen(false)}
      />
    </>
  )
}
```

### Async action with loading state

```tsx
export function AsyncSave() {
  const [open, setOpen] = useState(false)
  const [loading, setLoading] = useState(false)

  async function handleSave() {
    setLoading(true)
    await saveData()
    setLoading(false)
    setOpen(false)
  }

  return (
    <DialogWrapper
      open={open}
      onOpenChange={setOpen}
      trigger="Save"
      title="Save changes?"
      loading={loading}
      onAction={handleSave}
    />
  )
}
```

### Detached trigger

```tsx
import { Dialog as DialogPrimitive } from "@base-ui/react/dialog"
import { DialogTrigger, DialogWrapper } from "@/components/ui/dialog"

const handle = DialogPrimitive.createHandle()

<DialogTrigger handle={handle}>Open</DialogTrigger>

<DialogWrapper
  handle={handle}
  title="Detached dialog"
  description="Trigger is outside the wrapper."
/>
```

### Multiple triggers with payload

```tsx
const handle = DialogPrimitive.createHandle<{ name: string; role: string }>()

<DialogTrigger handle={handle} payload={{ name: 'Alice', role: 'Admin' }}>Alice</DialogTrigger>
<DialogTrigger handle={handle} payload={{ name: 'Bob', role: 'Editor' }}>Bob</DialogTrigger>

<DialogPrimitive.Root handle={handle}>
  {({ payload }) => (
    <DialogPrimitive.Portal>
      <DialogPrimitive.Popup>
        <DialogPrimitive.Title>{payload?.name}</DialogPrimitive.Title>
        <DialogPrimitive.Description>Role: {payload?.role}</DialogPrimitive.Description>
      </DialogPrimitive.Popup>
    </DialogPrimitive.Portal>
  )}
</DialogPrimitive.Root>
```

### Nested dialogs

```tsx
<DialogWrapper trigger="Open outer" title="Outer dialog" action="">
  <DialogWrapper
    trigger="Open inner"
    title="Inner dialog"
    description="Nested inside the outer dialog."
    action="Confirm"
  />
</DialogWrapper>
```

### Custom footer content

Set `action=""` and `cancel=""` to hide the default footer, then pass content via `children`:

```tsx
<DialogWrapper trigger="Open" title="Custom footer" action="" cancel="">
  <DialogFooter>
    <DialogClose render={<Button variant="outline" />}>Cancel</DialogClose>
    <Button onClick={handleAction}>Save</Button>
  </DialogFooter>
</DialogWrapper>
```

#### DialogFooterWrapper
<AutoTypeTable
name="DialogFooterWrapper"
type={`export interface DialogFooterWrapper {
  /**
   * Text or element for the cancel button. Set to \`""\` or \`null\` to hide.
   * @default "Cancel"
   */
  cancel?: React.ReactNode;

  /**
   * Text or element for the action button. Set to \`""\` or \`null\` to hide.
   */
  action?: React.ReactNode;

  /**
   * When true, disables both buttons and shows a spinner on the action button.
   * Dialog blocks close (Escape, backdrop) while loading.
   * @default false
   */
  loading?: boolean;

  /**
   * Additional CSS classes for the footer container.
   */
  footerCls?: string;

  /**
   * Additional CSS classes for the action button.
   */
  actionCls?: string;

  /**
   * Additional CSS classes for the cancel button.
   */
  cancelCls?: string;

  /**
   * Callback fired when the action button is clicked.
   * Dialog stays open — close manually via state if needed.
   */
  onAction?: () => void;

  /**
   * Callback fired when the cancel button is clicked.
   */
  onCancel?: () => void;
}`}

#### DialogWrapper
<AutoTypeTable
name="DialogWrapper"
type={`export interface DialogWrapper {
  /**
   * Content rendered inside the auto-wrapped trigger button.
   * Accepts strings, icons, or any ReactNode.
   * Omit to use a detached trigger or controlled open state.
   *
   * \`\`\`tsx
   * trigger="Open"
   * trigger={<><Settings /> Settings</>}
   * \`\`\`
   */
  trigger?: React.ReactNode;

  /**
   * CSS class applied to the trigger element.
   */
  triggerCls?: string;

  /**
   * Additional props for the trigger — use to pass handle, payload, id,
   * or a custom render element (for non-button triggers).
   *
   * \`\`\`tsx
   * triggerProps={{ render: <a href="#" />, nativeButton: false }}
   * triggerProps={{ handle: myHandle }}
   * \`\`\`
   */
  triggerProps?: Omit<DialogPrimitive.Trigger.Props, 'children' | 'className'>;

  /**
   * Dialog title.
   */
  title?: React.ReactNode;

  /**
   * Dialog description. Set to \`""\` to hide.
   */
  description?: React.ReactNode;

  /**
   * Additional content rendered between header and footer.
   */
  children?: React.ReactNode;

  /**
   * Whether to show the X close button in the top-right corner.
   * @default true
   */
  showCloseButton?: boolean;

  /**
   * Additional CSS classes for the dialog content panel.
   */
  contentCls?: string;

  /**
   * Additional CSS classes for the header section.
   */
  headerCls?: string;

  /**
   * Additional CSS classes for the title.
   */
  titleCls?: string;

  /**
   * Additional CSS classes for the description.
   */
  descriptionCls?: string;

  /**
   * ...otherProps: All DialogFooterWrapper props (cancel, action, loading,
   * footerCls, actionCls, cancelCls, onAction, onCancel) can be passed directly.
   * All Base UI Dialog.Root props (open, onOpenChange, handle, defaultOpen, etc.)
   * are also forwarded to the root.
   */
  otherProps?: never;
}`}
/>
