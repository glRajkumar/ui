## Usage

### Basic

```tsx
import { HoverCardWrapper } from "@/components/ui/hover-card"

<HoverCardWrapper
  trigger="@glrk_ui"
  triggerCls="px-4 py-1.5 border rounded text-sm"
  title="Glrk UI"
  description="Supporting description text."
/>
```

### Title and description

Pass `title` and `description` directly — renders as `HoverCardTitle` and `HoverCardDescription` inside a `HoverCardHeader`:

```tsx
<HoverCardWrapper
  trigger="Hover me"
  title="Card title"
  description="Supporting description text."
/>
```

For arbitrary content, use `content` instead — useful for things like a profile preview:

```tsx
<HoverCardWrapper
  trigger="View profile"
  content={
    <div className="flex items-center gap-3">
      <div className="size-9 rounded-full bg-foreground" />
      <div className="flex flex-col gap-0.5">
        <span className="text-sm font-medium">Alice Chen</span>
        <span className="text-xs text-muted-foreground">Product Designer</span>
      </div>
    </div>
  }
/>
```

### Delay

Opens on hover after `delay` ms, closes after `closeDelay` ms once the pointer leaves:

```tsx
<HoverCardWrapper
  trigger="Hover (delayed)"
  triggerProps={{ delay: 800, closeDelay: 500 }}
  title="Delayed card"
  description="800 ms open delay, 500 ms close grace."
/>
```

### Controlled

```tsx
const [open, setOpen] = React.useState(false)

<HoverCardWrapper
  open={open}
  onOpenChange={setOpen}
  trigger="Anchor"
  title="Controlled"
  description="Driven by external state."
/>
```

### Arrow

```tsx
<HoverCardWrapper
  trigger="Hover"
  contentProps={{ showArrow: true, sideOffset: 10 }}
  title="With arrow"
  description="Arrow points toward the trigger."
/>
```

Control placement with `side`:

```tsx
<HoverCardWrapper
  trigger="Top"
  contentProps={{ side: "top", showArrow: true, sideOffset: 10 }}
  title="side: top"
  description="Arrow points toward trigger."
/>
```

### Positioning

```tsx
<HoverCardWrapper trigger="Top"    contentProps={{ side: "top" }}    title="Top" />
<HoverCardWrapper trigger="Right"  contentProps={{ side: "right" }}  title="Right" />
<HoverCardWrapper trigger="Bottom" contentProps={{ side: "bottom" }} title="Bottom" />
<HoverCardWrapper trigger="Left"   contentProps={{ side: "left" }}   title="Left" />
```

   * \`\`\`tsx

   * \`\`\`

  triggerProps?: Omit<PreviewCardPrimitive.Trigger.Props, 'className' | 'children'>;

   * CSS classes for the popup element.

  contentCls?: string;

  contentProps?: Omit<hoverCardContentType, 'className'>;

   * ...rootProps: All Base UI PreviewCard.Root props forwarded to the root.
   * Includes: open, onOpenChange, defaultOpen, actionsRef, handle, etc.
   */
  otherProps?: never;
