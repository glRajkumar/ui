## Usage

### Standalone (InputGroupWrapper)

```tsx
import { InputGroupWrapper, InputGroupText } from "@/components/ui/input-group"

<InputGroupWrapper
  placeholder="username"
  addonStart={<InputGroupText>@</InputGroupText>}
/>
```

### With icon prefix

```tsx
import { SearchIcon } from "lucide-react"

<InputGroupWrapper
  placeholder="Search..."
  addonStart={<InputGroupText><SearchIcon /></InputGroupText>}
/>
```

### With text suffix

```tsx
<InputGroupWrapper
  placeholder="yoursite"
  addonEnd={<InputGroupText>.com</InputGroupText>}
/>
```

### With button addon

```tsx
import { CopyIcon } from "lucide-react"
import { InputGroupWrapper, InputGroupButton, InputGroupText } from "@/components/ui/input-group"

<InputGroupWrapper
  defaultValue="https://example.com"
  readOnly
  addonEnd={
    <InputGroupButton>
      <CopyIcon />
    </InputGroupButton>
  }
/>
```

### With field label and error (InputGroupWrapper)

`InputGroupWrapper` from `field-wrapper` wraps the input group inside a `Field` with label and error display:

```tsx
import { InputGroupWrapper } from "@/components/ui/field-wrapper"
import { InputGroupText } from "@/components/ui/input-group"
import { DollarSignIcon } from "lucide-react"

<InputGroupWrapper
  name="amount"
  label="Amount"
  placeholder="0.00"
  addonStart={<InputGroupText><DollarSignIcon /></InputGroupText>}
  addonEnd={<InputGroupText>USD</InputGroupText>}
/>
```

### With validation error

```tsx
<InputGroupWrapper
  name="username"
  label="Username"
  placeholder="username"
  addonStart={<InputGroupText>@</InputGroupText>}
  invalid
  error={{ message: "Username is already taken." }}
/>
```

### React Hook Form

```tsx
import { InputGroupWrapper } from "@/components/ui/field-wrapper-rhf"
import { InputGroupText } from "@/components/ui/input-group"

<InputGroupWrapper
  name="username"
  label="Username"
  control={form.control}
  addonStart={<InputGroupText>@</InputGroupText>}
  placeholder="username"
/>
```

### TanStack Form

```tsx
import { InputGroupText } from "@/components/ui/input-group"

<form.AppField name="username">
  {field => (
    <field.InputGroupField
      label="Username"
      placeholder="username"
      addonStart={<InputGroupText>@</InputGroupText>}
    />
  )}
</form.AppField>
```

### Addon alignment options

```tsx
// inline addons (default) — left/right of input on same line
<InputGroupAddon align="inline-start">...</InputGroupAddon>
<InputGroupAddon align="inline-end">...</InputGroupAddon>

// block addons — above/below the input (stacked layout)
<InputGroupAddon align="block-start">Character limit: 280</InputGroupAddon>
<InputGroupInput />
<InputGroupAddon align="block-end">0 / 280</InputGroupAddon>
```
