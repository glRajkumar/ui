## Usage

### Single checkbox — CheckboxWrapper

```tsx
import { CheckboxWrapper } from "@/components/ui/checkbox"

<CheckboxWrapper label="Accept terms and conditions" />
<CheckboxWrapper
  label="Email notifications"
  description="Receive updates about your account."
/>
<CheckboxWrapper label="Disabled" disabled />
```

### Group — CheckboxGroupWrapper

```tsx
import { CheckboxGroupWrapper } from "@/components/ui/checkbox-group"

<CheckboxGroupWrapper
  defaultValue={["email"]}
  items={[
    { value: "email", label: "Email", description: "Notify via email." },
    { value: "sms", label: "SMS" },
    { value: "push", label: "Push notifications" },
    { value: "locked", label: "Cannot change", disabled: true },
  ]}
/>
```

### Select-all parent checkbox

Set `parentLabel` to add a parent checkbox that shows indeterminate when partially selected:

```tsx
<CheckboxGroupWrapper
  defaultValue={["email"]}
  parentLabel="All notifications"
  items={[
    { value: "email", label: "Email" },
    { value: "sms", label: "SMS" },
    { value: "push", label: "Push" },
  ]}
/>
```

### Controlled

```tsx
const [value, setValue] = useState<string[]>(["email"])

<CheckboxGroupWrapper
  value={value}
  onValueChange={(v) => setValue(v)}
  parentLabel="Select all"
  items={[
    { value: "email", label: "Email" },
    { value: "sms", label: "SMS" },
  ]}
/>
```

### Horizontal orientation

```tsx
<CheckboxGroupWrapper
  orientation="horizontal"
  defaultValue={["react"]}
  items={[
    { value: "react", label: "React" },
    { value: "vue", label: "Vue" },
    { value: "svelte", label: "Svelte" },
  ]}
/>
```
