## Usage

### Basic
```tsx
import { PaginationWrapper } from "../ui/pagination";

export function Basic() {
  return (
    <PaginationWrapper
      total={20}
    />
  )
}
```

### Controlled
```tsx
import { useState } from "react"
import { PaginationWrapper } from "../ui/pagination";

export function Controlled() {
  const [value, setValue] = useState(5)

  return (
    <PaginationWrapper
      total={20}
      page={value}
      onPageChange={setValue}
    />
  )
}
```

   *

   * \`\`\`tsx
   * const [currentPage, setCurrentPage] = useState(1);

   *   total={20}
   *   page={currentPage}
   *   onPageChange={setCurrentPage}
   * />
   * \`\`\`
   * 
   * ## With Custom Sibling Count:
   * \`\`\`tsx

   * <PaginationWrapper
   *   total={50}
   *   siblingCount={2}
   *   onPageChange={handlePageChange}
   * />
   * 
   * // Compact version with fewer visible pages
   * <PaginationWrapper
   *   total={50}
   *   siblingCount={0}
   *   onPageChange={handlePageChange}
   * />
   * \`\`\`
   */

  /**
   * Total number of pages
   * 
   * Must be a positive integer
   * 
   * \`\`\`tsx
   * total={1}  // Single page (no navigation needed)
   * total={10} // 10 pages
   * total={100} // 100 pages with ellipsis
   * \`\`\`
   */
  total: number;

  /**
   * Current active page (controlled mode)
   * 
   * When provided, component operates in controlled mode
   * Must be used with \`onPageChange\` to update the value
   * 
   * \`\`\`tsx
   * const [page, setPage] = useState(1);
   * 
   * <PaginationWrapper
   *   total={20}
   *   page={page}
   *   onPageChange={setPage}
   * />
   * \`\`\`
   * 
   * @default 1 (uncontrolled)
   */
  page?: number;

  /**
   * Number of sibling pages to show on each side of current page
   * 
   * Controls how many page numbers appear before and after the current page
   * Higher values show more pages but take up more space
   * 
   * ## Examples:
   * 
   * **siblingCount={0}** (current page only):
   * \`1 ... 5 ... 10\`
   * 
   * **siblingCount={1}** (default):
   * \`1 ... 4 5 6 ... 10\`
   * 
   * **siblingCount={2}**:
   * \`1 ... 3 4 5 6 7 ... 10\`
   * 
   * \`\`\`tsx
   * siblingCount={0} // Minimal: only current page

  siblingCount?: number;
