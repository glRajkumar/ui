## Usage

### Basic

```tsx
import { AutocompleteWrapper } from "@/components/ui/autocomplete"

<AutocompleteWrapper
  items={["Apple", "Banana", "Orange"]}
  placeholder="Search fruits..."
  className="w-64"
/>
```

### With clear and trigger buttons

```tsx
<AutocompleteWrapper
  items={items}
  placeholder="Search..."
  showClear
  showTrigger
  className="w-64"
/>
```

### Controlled

```tsx
const [value, setValue] = useState("")

<AutocompleteWrapper
  items={items}
  value={value}
  onValueChange={setValue}
  placeholder="Search..."
  showClear
/>
```

### Option objects

```tsx
const frameworks = [
  { value: "next", label: "Next.js" },
  { value: "remix", label: "Remix" },
  { value: "astro", label: "Astro" },
]

<AutocompleteWrapper items={frameworks} placeholder="Select framework..." />
```

### Grouped options

```tsx
const grouped = [
  {
    group: "Frontend",
    options: [
      { value: "react", label: "React" },
      { value: "vue", label: "Vue" },
    ],
  },
  {
    group: "Backend",
    options: [
      { value: "node", label: "Node.js" },
      { value: "go", label: "Go" },
    ],
  },
]

<AutocompleteWrapper items={grouped} placeholder="Search tech..." showClear />
```

### Limit results

```tsx
<AutocompleteWrapper items={items} placeholder="Search..." limit={5} />
```

### Inline mode

Completes input text inline as the user types:

```tsx
<AutocompleteWrapper items={items} mode="inline" placeholder="Start typing..." />
```

### Async search

Disable client-side filter with `filter={null}` and supply results externally:

```tsx
const [inputValue, setInputValue] = useState("")
const [results, setResults] = useState<string[]>([])
const [loading, setLoading] = useState(false)

async function handleValueChange(val: string) {
  setInputValue(val)
  setLoading(true)
  const data = await fetchSuggestions(val)
  setResults(data)
  setLoading(false)
}

<AutocompleteWrapper
  items={results}
  value={inputValue}
  onValueChange={handleValueChange}
  filter={null}
  isLoading={loading}
  placeholder="Search..."
  showClear
/>
```

### Using primitives

```tsx
import {
  AutocompleteRoot,
  AutocompleteInput,
  AutocompleteContent,
  AutocompleteList,
  AutocompleteItem,
  AutocompleteEmpty,
} from "@/components/ui/autocomplete"

<AutocompleteRoot items={["Apple", "Banana", "Orange"]}>
  <AutocompleteInput placeholder="Search..." showClear />
  <AutocompleteContent>
    <AutocompleteEmpty>No results</AutocompleteEmpty>
    <AutocompleteList>
      {(item: unknown) => (
        <AutocompleteItem key={item as string} value={item as string}>
          {item as string}
        </AutocompleteItem>
      )}
    </AutocompleteList>
  </AutocompleteContent>
</AutocompleteRoot>
```

   */
  showClear?: boolean;

  showTrigger?: boolean;

  /**

  contentCls?: string;

   * CSS class applied to each suggestion item.

  // autoHighlight, limit, filter, itemToStringValue,
