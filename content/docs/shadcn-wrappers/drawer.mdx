## Usage

### Basic

```tsx
import { DrawerWrapper } from "@/components/ui/drawer"

export function Basic() {
  return (
    <DrawerWrapper
      trigger="Open"
      triggerCls={buttonVariants({ variant: 'outline' })}
      title="Basic drawer"
      description="Swipe down to dismiss."
      action="Confirm"
    />
  )
}
```

### Sides

Use `side` to set which edge the drawer slides from. Pair with `swipeDirection` to match:

```tsx
<DrawerWrapper side="bottom" swipeDirection="down" trigger="Bottom" title="Bottom drawer" />
<DrawerWrapper side="top"    swipeDirection="up"   trigger="Top"    title="Top drawer" />
<DrawerWrapper side="left"   swipeDirection="left" trigger="Left"   title="Left drawer" />
<DrawerWrapper side="right"  swipeDirection="right" trigger="Right" title="Right drawer" />
```

### Snap points

```tsx
<DrawerWrapper
  trigger="Snap Points"
  title="Snap points"
  description="Drawer snaps to 40% or 100% height. Drag to switch."
  snapPoints={[0.4, 1]}
  action="Confirm"
/>
```

### Non-modal

Page content stays interactive while the drawer is open:

```tsx
<DrawerWrapper
  trigger="Non-modal"
  modal={false}
  title="Non-modal drawer"
  description="The page behind remains interactive."
  action="Close"
/>
```

### Nested drawers

```tsx
<DrawerWrapper trigger="Open outer" title="Outer drawer" action="">
  <DrawerWrapper
    trigger="Open inner"
    title="Inner drawer"
    description="Nested inside the outer drawer."
    action="Confirm"
  />
</DrawerWrapper>
```

### Controlled

```tsx
export function Controlled() {
  const [open, setOpen] = useState(false)

  return (
    <>
      <button onClick={() => setOpen(true)}>Open</button>

      <DrawerWrapper
        open={open}
        onOpenChange={setOpen}
        title="Controlled drawer"
        action="Confirm"
        onAction={() => setOpen(false)}
      />
    </>
  )
}
```

### Async action with loading state

```tsx
export function AsyncSubmit() {
  const [open, setOpen] = useState(false)
  const [loading, setLoading] = useState(false)

  async function handleSubmit() {
    setLoading(true)
    await submitData()
    setLoading(false)
    setOpen(false)
  }

  return (
    <DrawerWrapper
      open={open}
      onOpenChange={setOpen}
      trigger="Submit"
      title="Submit form?"
      action="Submit"
      loading={loading}
      onAction={handleSubmit}
    />
  )
}
```

### Custom footer content

```tsx
<DrawerWrapper trigger="Open" title="Custom footer" action="" cancel="">
  <DrawerFooter>
    <DrawerClose className={buttonVariants({ variant: 'outline' })}>Cancel</DrawerClose>
    <Button onClick={handleAction}>Confirm</Button>
  </DrawerFooter>
</DrawerWrapper>
```
