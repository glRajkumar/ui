## Usage

### Basic
```tsx
import { ItemWrapper } from "@/components/ui/item"
import { Button } from "@/components/ui/button"

export function Basic() {
  return (
    <ItemWrapper
      header="Product"
      title="Wireless Headphones"
      description="Noise-cancelling over-ear headphones with long battery life."
      content="Perfect for work, travel, and everyday listening. Supports fast charging."
      actions={<Button size="sm">Buy Now</Button>}
      footer="Updated 2 days ago"
    />
  )
}
```

### Item Group
```tsx
import { type itemsT, ItemGroupWrapper } from "@/components/ui/item"
import { Button } from "@/components/ui/button"

export function Group() {
   const items: itemsT = [
    {
      header: "Product",
      title: "Wireless Headphones",
      description: "Noise-cancelling over-ear headphones with long battery life.",
      content: "Perfect for work, travel, and everyday listening. Supports fast charging.",
      actions: <Button size="sm">Buy Now</Button>,
      footer: "Updated 2 days ago",
      itemProps: { variant: "outline" }
    },
    {
      title: "Smart Watch",
      description: "Track your fitness, heart rate and notifications.",
      actions: <Button variant="outline" size="sm">View Details</Button>,
      footer: "In Stock",
      itemProps: { variant: "muted" }
    },
  ]

  return (
    <ItemGroupWrapper
      items={items}
    />
  )
}
```

#### ItemWrapper
<AutoTypeTable 
name="ItemWrapper" 
type={`export interface ItemWrapper {
  /**
   * Custom header content
   * 
   * Displayed at the top of the item, before title
   * Useful for badges, timestamps, or metadata
   * 
   * \`\`\`tsx
   * header={<Badge variant="secondary">Draft</Badge>}

   *   <div className="flex justify-between">

   *     <span className="text-xs">5 min ago</span>
   *   </div>
   * }
   * \`\`\`
   */
  header?: React.ReactNode;

  /**
   * Main title text
   * 
   * Primary heading for the item
   * 
   * \`\`\`tsx
   * title="Task Name"
   * title={<h3 className="font-bold">Custom Title</h3>}
   * \`\`\`
   */
  title?: React.ReactNode;

  /**
   * Media element (avatar, icon, image, thumbnail)
   * 
   * Typically displayed on the left side or at the top
   * 
   * \`\`\`tsx
   * media={<Avatar src="/avatar.jpg" />}
   * media={<FileIcon className="w-8 h-8" />}
   * media={<img src="/thumbnail.jpg" alt="" className="w-16 h-16 rounded" />}
   * \`\`\`
   */
  media?: React.ReactNode;

  /**
   * Description or subtitle text
   * 
   * Secondary text providing additional context
   * 
   * \`\`\`tsx
   * description="This is a brief description"
   * description={
   *   <span className="text-sm text-muted-foreground">
   *     Last updated today
   *   </span>
   * }
   * \`\`\`
   */
  description?: React.ReactNode;

  /**
   * Action buttons or interactive elements
   * 
   * Typically displayed on the right side or at the bottom
   * 
   * \`\`\`tsx
   * actions={<Button size="sm">View Details</Button>}
   * actions={
   *   <div className="flex gap-2">
   *     <Button size="icon" variant="ghost"><EditIcon /></Button>
   *     <Button size="icon" variant="ghost"><TrashIcon /></Button>
   *   </div>
   * }
   * \`\`\`
   */
  actions?: React.ReactNode;

  /**
   * Additional custom content
   * 
   * Flexible area for any extra content between description and actions
   * 
   * \`\`\`tsx
   * content={
   *   <div className="mt-3 flex gap-2">
   *     <Tag>React</Tag>
   *     <Tag>TypeScript</Tag>
   *   </div>
   * }
   * \`\`\`
   */
  content?: React.ReactNode;

  /**
   * Footer content
   * 
   * Displayed at the bottom of the item
   * Useful for timestamps, metadata, or additional actions
   * 
   * \`\`\`tsx
   * footer={<span className="text-xs text-muted-foreground">Created 2 days ago</span>}
   * footer={
   *   <div className="flex justify-between items-center">
   *     <span>3 comments</span>
   *     <Button variant="link" size="sm">Reply</Button>
   *   </div>
   * }
   * \`\`\`
   */
  footer?: React.ReactNode;

  /**
   * Additional CSS classes for the root Item container
   * 
   * \`\`\`tsx
   * itemWrapperCls="hover:bg-muted/50 transition-colors"
   * \`\`\`
   */
  itemWrapperCls?: string;

  /**
   * Additional CSS classes for the ItemHeader section
   * 
   * \`\`\`tsx
   * headerCls="mb-2"
   * \`\`\`
   */
  headerCls?: string;

  /**
   * Additional CSS classes for the ItemTitle element
   * 
   * \`\`\`tsx
   * titleCls="text-lg font-semibold"
   * \`\`\`
   */
  titleCls?: string;

  /**
   * Additional CSS classes for the ItemMedia container
   * 
   * \`\`\`tsx
   * mediaCls="rounded-full overflow-hidden"
   * \`\`\`
   */
  mediaCls?: string;

  /**
   * Additional CSS classes for the ItemDescription element
   * 
   * \`\`\`tsx
   * descriptionCls="text-sm text-muted-foreground line-clamp-2"
   * \`\`\`
   */
  descriptionCls?: string;

  /**
   * Additional CSS classes for the ItemContent section
   * 
   * Contains title, description, and custom content
   * 
   * \`\`\`tsx
   * contentCls="flex-1"
   * \`\`\`
   */
  contentCls?: string;

  /**
   * Additional CSS classes for the ItemActions section
   * 
   * \`\`\`tsx
   * actionsCls="ml-auto"
   * \`\`\`
   */
  actionsCls?: string;

  /**
   * Additional CSS classes for the ItemFooter section
   * 
   * \`\`\`tsx
   * footerCls="pt-3 border-t"
   * \`\`\`
   */
  footerCls?: string;

  /**
   * Additional props passed to the root Item component
   * 
   * Excludes className (use itemWrapperCls instead)
   * 
   * \`\`\`tsx
   * itemProps={{
   *   onClick: () => console.log('clicked'),
   *   role: "button",
   *   tabIndex: 0
   * }}
   * \`\`\`
   */
  itemProps?: Omit<React.HTMLAttributes<HTMLDivElement>, "className">;

  /**
   * Additional props passed to the ItemMedia component
   * 
   * Excludes className (use mediaCls instead)
   * 
   * \`\`\`tsx
   * itemMediaProps={{
   *   variant: "circular",
   *   size: "lg"
   * }}
   * \`\`\`
   */
  itemMediaProps?: Omit<React.ComponentProps<"div"> & VariantProps<typeof itemMediaVariants>, "className">;
}`}
