## Usage

### Basic — Toaster wrapper

Wrap your app (or subtree) with `<Toaster>`. Call `useToastManager().add()` from any child component to fire a toast.

```tsx
import { Toaster, useToastManager } from "@/components/ui/toast";

function TriggerButton() {
  const { add } = useToastManager();
  return (
    <button onClick={() => add({ title: "Saved", description: "Changes saved." })}>
      Save
    </button>
  );
}

export function App() {
  return (
    <Toaster>
      <TriggerButton />
    </Toaster>
  );
}
```

### Global manager

Create a manager instance outside the React tree so you can fire toasts from anywhere (e.g. axios interceptors, utility functions).

```tsx
import { createToastManager, Toaster } from "@/components/ui/toast";

export const toast = createToastManager();

// In your root layout
<Toaster toastManager={toast} />

// Anywhere in the app — no hook needed
toast.add({ title: "Uploaded", description: "File upload complete." });
```

### Composing primitives

Use the individual primitives for full control.

```tsx
import {
  ToastProvider,
  ToastPortal,
  ToastViewport,
  ToastRoot,
  ToastContent,
  ToastTitle,
  ToastDescription,
  ToastClose,
  useToastManager,
} from "@/components/ui/toast";

function ToastList() {
  const { toasts } = useToastManager();
  return toasts.map((toast) => (
    <ToastRoot key={toast.id} toast={toast}>
      <ToastContent>
        <ToastTitle />
        <ToastDescription />
        <ToastClose />
      </ToastContent>
    </ToastRoot>
  ));
}

export function App() {
  return (
    <ToastProvider>
      {/* app content */}
      <ToastPortal>
        <ToastViewport>
          <ToastList />
        </ToastViewport>
      </ToastPortal>
    </ToastProvider>
  );
}
```
