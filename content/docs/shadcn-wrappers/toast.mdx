## Usage

### Basic

Wrap your app (or subtree) with `<Toaster>`. Use `useToast()` from any child component.

```tsx
import { Toaster, useToast } from "@/components/ui/toast";

function SaveButton() {
  const toast = useToast();
  return (
    <button onClick={() => toast.success("Changes saved.")}>Save</button>
  );
}

export function App() {
  return (
    <Toaster>
      <SaveButton />
    </Toaster>
  );
}
```

### Variants

```tsx
const toast = useToast();

toast.add({ description: "Notification sent." });          // default
toast.success("Changes saved.");
toast.error("Something went wrong.", { title: "Error" });
toast.warning("Session expires soon.");
toast.info("Update available.", { title: "v2.0" });
```

### Promise

Automatically transitions loading → success or error when the promise resolves.

```tsx
toast.promise(uploadFile(file), {
  loading: "Uploading...",
  success: (data) => `${data.name} uploaded.`,
  error: "Upload failed.",
});
```

### Action button

Pass `actionProps` to render a button inside the toast.

```tsx
toast.add({
  title: "File deleted",
  description: "document.pdf removed.",
  actionProps: {
    children: "Undo",
    onClick: () => restoreFile(),
  },
});
```

### Global manager

Create a manager outside React to trigger toasts from anywhere (axios interceptors, utils, etc.).

```tsx
import { createToastManager, Toaster } from "@/components/ui/toast";

export const toast = createToastManager();

// Root layout
<Toaster toastManager={toast} />

// Anywhere — no hook needed
toast.add({ description: "Done.", type: "success" });
toast.promise(fetch("/api/data"), {
  loading: { description: "Loading...", type: "loading" },
  success: { description: "Loaded.", type: "success" },
  error: { description: "Failed.", type: "error" },
});
```

### Composing primitives

```tsx
import {
  ToastProvider,
  ToastPortal,
  ToastViewport,
  ToastRoot,
  ToastContent,
  ToastTitle,
  ToastDescription,
  ToastAction,
  ToastClose,
  useToastManager,
} from "@/components/ui/toast";

function ToastList() {
  const { toasts } = useToastManager();
  return toasts.map((toast) => (
    <ToastRoot key={toast.id} toast={toast}>
      <ToastContent>
        <ToastTitle />
        <ToastDescription />
        {toast.actionProps && <ToastAction {...toast.actionProps} />}
        <ToastClose />
      </ToastContent>
    </ToastRoot>
  ));
}
```

## Reference

### useToast

```ts
const toast = useToast();

toast.add(options)                          // → string (toast id)
toast.success(description, options?)        // → string
toast.error(description, options?)          // → string
toast.warning(description, options?)        // → string
toast.info(description, options?)           // → string
toast.promise(promise, messages)            // → Promise<Value>
toast.update(id, partialOptions)
toast.close(id?)
```

### toastTypeT

```ts
type toastTypeT = "default" | "success" | "error" | "warning" | "info" | "loading"
```

Each type renders a colored border/background and a corresponding icon.
