## Usage

### Complex
```tsx
import { CircleHelpIcon } from "lucide-react"
import { NavigationMenuWrapper, NavLinkItem, NavList } from "@/components/ui/navigation-menu-wrapper"

export function Complex() {
  const dropdownOptions = ["Option 1", "Option 2", "Option 3"]

  return (
    <NavigationMenuWrapper
      items={[
        {
          key: "home",
          trigger: "Home",
          content: (
            <div className="p-2 w-[250px]">
              <NavList
                items={[
                  { href: "/", children: "Dashboard" },
                  { href: "/about", children: "About Us" },
                ]}
              />
            </div>
          ),
        }, {
          key: "pricing",
          href: "/pricing",
          children: "Pricing",
        },
        {
          key: "features",
          trigger: "Features",
          content: (
            <div className="p-2 w-[300px]">
              <div className="text-lg font-semibold">Product Features</div>

              <NavList
                items={[
                  { href: "/fast", children: "Fast Performance" },
                  { href: "/secure", children: "Secure Infrastructure" },
                  {
                    href: "/todo",
                    children: <><CircleHelpIcon className="size-4" /> Backlog</>,
                  },
                ]}
              />

              <NavLinkItem href="/dashbard" >
                Dashboard
              </NavLinkItem>
            </div>
          ),
        }
      ]}
    />
  )
}
```

## Reference

#### Component Hierarchy

```bash
NavigationMenuWrapper (Main component)
  └─ NavigationMenuList
      └─ NavigationMenuItem (repeated for each item)
          ├─ NavLinkItem (for type="link")
          │   └─ Next.js Link
          │
          └─ NavigationMenuTrigger + Content (for type="trigger")
              ├─ Trigger element
              └─ NavigationMenuContent
                  └─ Custom content (can include NavList)

NavList (Standalone component)
  └─ <ul>
      └─ <li> (repeated for each item)
          └─ NavLinkItem
              └─ Next.js Link
```

#### NavigationMenuItem
##### Common
<AutoTypeTable 
name="NavigationMenuItem" 
type={`export interface NavigationMenuItem {
  /**
   * Unique identifier for the menu item
   * Required for React key prop
   */
  key: string;

  /**
   * Additional props passed to the NavigationMenuItem component
   * 
   * \`\`\`tsx
   * itemProps={{
   *   className: "custom-item-class",
   *   onMouseEnter: () => console.log("hover")
   * }}
   * \`\`\`
   */
  itemProps?: React.ComponentProps<typeof NavigationMenuItem>;

  /**
   * Type of navigation item
   * 
   * ## Link Type
   * Simple navigation link without dropdown
   * 
   * \`\`\`tsx
   * {
   *   key: "home",
   *   href: "/",
   *   children: "Home"
   * }
   * \`\`\`
   * 
   * ## Trigger Type  
   * Navigation item with dropdown content
   * 
   * \`\`\`tsx
   * {
   *   key: "products",
   *   trigger: "Products",
   *   content: <div>Product list here</div>,
   *   contentProps: {
   *     className: "w-[400px]"
   *   }
   * }
   * \`\`\`
   */
   item?: {
      trigger: React.ReactNode
      content: React.ReactNode
      triggerCls?: string
      contentCls?: string
      contentProps?: React.ComponentProps<typeof NavigationMenuContent>
    } | navLinkItemT
}`}
/>

##### Trigger-Content
<AutoTypeTable 
name="NavigationMenuItem" 
type={`export interface NavigationMenuItem {
  /**
   * // When type is "trigger"
   * Trigger element for dropdown menu
   * Only applicable when \`type="trigger"\`
   * 
   * Can be a string or custom React component

   * Can contain any React elements - lists, grids, custom layouts

   */
  content?: React.ReactNode;

   * Optional CSS classes for this specific accordion trigger

  triggerCls?: string;

  /**
   * Additional CSS classes for the content

  contentCls?: string;
