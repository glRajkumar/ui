```tsx title="ui/accordion.tsx"
type accordionItemT = {
  value: string
  trigger: React.ReactNode
  content: React.ReactNode
  className?: string
  triggerCls?: string
  contentCls?: string
  disabled?: boolean
}

type accordionItemsT = accordionItemT[]

type accordionWrapperProps = {
  items: accordionItemsT
  itemCls?: string
  triggerCls?: string
  contentCls?: string
  collapsible?: boolean
} & Omit<React.ComponentProps<typeof AccordionPrimitive.Root>, "type" | "collapsible">

function AccordionWrapper({
  items,
  itemCls,
  triggerCls,
  contentCls,
  ...props
}: accordionWrapperProps) {
  return (
    <Accordion {...(props)}>
      {items.map((item) => (
        <AccordionItem
          key={item.value}
          value={item.value}
          className={cn(itemCls, item.className)}
          disabled={item.disabled}
        >
          <AccordionTrigger className={cn("items-center justify-start gap-2", triggerCls, item.triggerCls)}>
            {item.trigger}
          </AccordionTrigger>

          <AccordionContent className={cn(contentCls, item.contentCls)}>
            {item.content}
          </AccordionContent>
        </AccordionItem>
      ))}
    </Accordion>
  )
}
```

Update following changes in `AccordionTrigger` component.

```tsx
function AccordionTrigger({
  className,
  children,
  ...props
}: React.ComponentProps<typeof AccordionPrimitive.Trigger>) {
  return (
    <AccordionPrimitive.Header className="flex">
      <AccordionPrimitive.Trigger
        data-slot="accordion-trigger"
        className={cn(
          // [!code ++] [!code word:>cursor-pointer]
          "group/accordion-trigger relative flex flex-1 items-start justify-between rounded-lg border border-transparent py-2.5 text-left text-sm font-medium transition-all outline-none hover:underline focus-visible:border-ring focus-visible:ring-3 focus-visible:ring-ring/50 focus-visible:after:border-ring aria-disabled:pointer-events-none aria-disabled:opacity-50 **:data-[slot=accordion-trigger-icon]:ml-auto **:data-[slot=accordion-trigger-icon]:size-4 **:data-[slot=accordion-trigger-icon]:text-muted-foreground cursor-pointer",
          className
        )}
        {...props}
      >
        {children}
         // [!code ++] [!code word:transition-transform duration-300 group-aria-expanded/accordion-trigger:rotate-180]
        <ChevronDownIcon data-slot="accordion-trigger-icon" className="pointer-events-none shrink-0 transition-transform duration-300 group-aria-expanded/accordion-trigger:rotate-180" />
      </AccordionPrimitive.Trigger>
    </AccordionPrimitive.Header>
  )
}

function AccordionContent({
  className,
  children,
  ...props
}: AccordionPrimitive.Panel.Props) {
  return (
    <AccordionPrimitive.Panel
      data-slot="accordion-content"
      // [!code ++] [!code word:h-(--accordion-panel-height) transition-[height] ease-out data-ending-style:h-0 data-starting-style:h-0]
      className="overflow-hidden text-sm h-(--accordion-panel-height) transition-[height] ease-out data-ending-style:h-0 data-starting-style:h-0"
      {...props}
    >
      <div
        className={cn(
          // [!code --] [!code word:h-(--accordion-panel-height) transition-[height] ease-out data-ending-style:h-0 data-starting-style:h-0]
          "pt-0 pb-2.5 [&_a]:underline [&_a]:underline-offset-3 [&_a]:hover:text-foreground [&_p:not(:last-child)]:mb-4",
          className
        )}
      >
        {children}
      </div>
    </AccordionPrimitive.Panel>
  )
}
```

## Usage

### Basic
```tsx
import { AccordionWrapper } from "@/components/ui/accordion";

export function Basic() {
  return (
    <AccordionWrapper
      items={[
       { value: "item 1", trigger: "Item 1", content: "Item 1 content" },
       { value: "item 2", trigger: "Item 2", content: "Item 2 content" },
       { value: "item 3", trigger: <><Apple /> Item 2</>, content: <>Item 2 content</> },
      ]}
    />
  )
}
```

## Reference

#### accordionItemT
<AutoTypeTable 
name="accordionItemT" 
type={`export type accordionItemT = {
  /**
   * Unique identifier for the accordion item
   * Used for controlling which items are open/closed
   */
  value: string;

  /**
   * Content displayed in the clickable trigger button
   * Can be a string or any React element
   */
  trigger: React.ReactNode;

  /**
   * Content displayed when the accordion item is expanded
   * Can be a string or any React element
   */
  content: React.ReactNode;

  /**
   * Optional CSS classes for this specific accordion item
   */
  className?: string;

  /**
   * Optional CSS classes for this specific accordion trigger
   */
  triggerCls?: string;

  /**
   * Optional CSS classes for this specific accordion content panel
   */
  contentCls?: string;

  /**
   * Whether this accordion item is disabled
   * Disabled items cannot be opened or closed
   */
  disabled?: boolean;
}`} 
/>

#### AccordionWrapper
<AutoTypeTable 
name="AccordionWrapper" 
type={`export interface AccordionWrapper {
  /**
   * \`accordionItemsT\` = \`accordionItemT[]\`
   * 
   * Array of accordion items to render
   * Each item contains the value, trigger, content, and optional styling
   * 
   * \`\`\`tsx
   * const items = [
   *   {
   *     value: "item-1",
   *     trigger: "What is React?",
   *     content: "React is a JavaScript library...",
   *   },
   *   {
   *     value: "item-2",
   *     trigger: <><Icon />Custom Trigger</>,
   *     content: <div>Rich content here</div>,
   *     disabled: true,
   *   }
   * ]
   * \`\`\`
   */
  items: accordionItemsT;

  /**
   * Additional CSS classes applied to all accordion items
   * Individual items can override this with their own \`className\`
   * 
   * \`\`\`tsx
   * itemCls="border-b"
   * \`\`\`
   */
  itemCls?: string;

  /**
   * Additional CSS classes applied to all accordion triggers
   * Individual items can override this with their own \`triggerCls\`
   * 
   * Note: Default classes include flex alignment and arrow positioning
   * 
   * \`\`\`tsx
   * triggerCls="font-semibold text-lg"
   * \`\`\`
   */
  triggerCls?: string;

  /**
   * Additional CSS classes applied to all accordion content panels
   * Individual items can override this with their own \`contentCls\`
   * 
   * \`\`\`tsx
   * contentCls="text-gray-600 pt-4"
   * \`\`\`
   */
  contentCls?: string;

  /**
   * ...otherProps: All properties from Base UI Accordion Root can be passed directly.
   */
  otherProps: AccordionPrimitive.Root
}`}
