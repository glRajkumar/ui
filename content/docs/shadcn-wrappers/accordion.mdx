## Usage

### Basic

```tsx
import { AccordionWrapper } from "@/components/ui/accordion"

<AccordionWrapper
  items={[
    { value: "q1", trigger: "Is it accessible?", content: "Yes. Adheres to WAI-ARIA." },
    { value: "q2", trigger: "Is it styled?", content: "Yes. Matches other components." },
  ]}
/>
```

### Multiple open panels

```tsx
<AccordionWrapper items={items} multiple />
```

### Indicator placement

```tsx
<AccordionWrapper items={items} indicatorAt="left" />   // chevron left
<AccordionWrapper items={items} indicatorAt="right" />  // default
```

Per-item override:

```tsx
<AccordionWrapper
  items={[
    { value: "a", trigger: "Right indicator", content: "..." },
    { value: "b", trigger: "Left indicator", content: "...", indicatorAt: "left" },
  ]}
/>
```

### Default open

```tsx
<AccordionWrapper items={items} defaultValue={["q1"]} />
```

### Controlled

```tsx
const [value, setValue] = useState<string[]>(["q1"])

<AccordionWrapper
  items={items}
  value={value}
  onValueChange={setValue}
  multiple
/>
```

### Disabled

```tsx
// Disable entire accordion
<AccordionWrapper items={items} disabled />

// Disable specific item
<AccordionWrapper
  items={[
    { value: "a", trigger: "Enabled", content: "..." },
    { value: "b", trigger: "Disabled", content: "...", itemProps: { disabled: true } },
  ]}
/>
```

### Keep panels mounted

Preserves DOM when closed — useful for forms or avoiding re-mount cost:

```tsx
<AccordionWrapper items={items} contentProps={{ keepMounted: true }} />
```

### Hidden until found

Enables browser Ctrl+F search inside closed panels:

```tsx
<AccordionWrapper items={items} contentProps={{ hiddenUntilFound: true }} />
```

### Custom heading level

Default heading is `<h3>`. Use `triggerProps.headerProps.render` to change:

```tsx
// All items use h2
<AccordionWrapper items={items} triggerProps={{ headerProps: { render: <h2 /> } }} />

// Per-item h1
<AccordionWrapper
  items={[
    { value: "a", trigger: "Page title", content: "...", triggerProps: { headerProps: { render: <h1 /> } } },
  ]}
/>
```

### Global vs per-item overrides

Global props apply to all items. Per-item props override globals:

```tsx
<AccordionWrapper
  items={[
    { value: "a", trigger: "Global style", content: "..." },
    { value: "b", trigger: "Custom style", content: "...", triggerCls: "text-primary" },
  ]}
  triggerCls="font-semibold"
  indicatorAt="left"
  contentProps={{ keepMounted: true }}
/>
```

### Using primitives

```tsx
import { Accordion, AccordionItem, AccordionTrigger, AccordionContent, AccordionHeader } from "@/components/ui/accordion"

<Accordion>
  <AccordionItem value="a">
    <AccordionTrigger indicatorAt="left" headerProps={{ render: <h2 /> }}>
      Title
    </AccordionTrigger>
    <AccordionContent keepMounted>Content</AccordionContent>
  </AccordionItem>
</Accordion>
```

## Reference

#### accordionSharedT
<AutoTypeTable
name="accordionSharedT"
type={`export type accordionSharedT = {
  /**
   * Additional CSS classes for trigger buttons.
   */
  triggerCls?: string;

  /**
   * Additional CSS classes for content panels.
   */
  contentCls?: string;

  /**
   * Placement of the chevron indicator.
   * @default "right"
   */
  indicatorAt?: 'right' | 'left';

  /**
   * Props forwarded to AccordionItem — all Accordion.Item.Props except value and className.
   * Use for: disabled, render, etc.
   */
  itemProps?: Omit<AccordionPrimitive.Item.Props, 'value' | 'className'>;

  /**
   * Props forwarded to AccordionTrigger — all Accordion.Trigger.Props except className and children.
   * Also accepts headerProps to customise the heading element (render prop for heading level).
   *
   * \`\`\`tsx
   * triggerProps={{ headerProps: { render: <h2 /> } }}
   * \`\`\`
   */
  triggerProps?: Omit<AccordionPrimitive.Trigger.Props, 'className' | 'children'> & {
    headerProps?: AccordionPrimitive.Header.Props;
  };

  /**
   * Props forwarded to AccordionContent (Panel) — all Accordion.Panel.Props except className and children.
   * Use for: keepMounted, hiddenUntilFound, render, etc.
   *
   * \`\`\`tsx
   * contentProps={{ keepMounted: true }}
   * contentProps={{ hiddenUntilFound: true }}
   * \`\`\`
   */
  contentProps?: Omit<AccordionPrimitive.Panel.Props, 'className' | 'children'>;
}`}
/>

#### accordionItemT
<AutoTypeTable
name="accordionItemT"
type={`export type accordionItemT = {
  /**
   * Unique identifier — used for controlled value and defaultValue.
   */
  value: string;

  /**
   * Content inside the trigger button. String or any ReactNode.
   */
  trigger: React.ReactNode;

  /**
   * Content shown when expanded. String or any ReactNode.
   */
  content: React.ReactNode;

  /**
   * CSS class for this item's wrapper element.
   */
  className?: string;
} & accordionSharedT;`}
/>

All `accordionSharedT` fields (`triggerCls`, `contentCls`, `indicatorAt`, `itemProps`, `triggerProps`, `contentProps`) can be set per-item to override globals.
