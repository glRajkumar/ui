## Usage

### Basic

```tsx
import { type menubaritemsT, MenubarWrapper } from "@/components/ui/menubar-wrapper"

const opts: menubaritemsT = [
  { key: "file", trigger: "File", items: ["New", "Open", "Save"] },
  { key: "edit", trigger: "Edit", items: ["Cut", "Copy", "Paste"] },
]

<MenubarWrapper items={opts} />
```

### Checkbox

```tsx
import { type menubarCheckboxitemsT, MenubarCheckboxWrapper } from "@/components/ui/menubar-wrapper"

const [checked, setChecked] = useState<allowedPrimitiveT[]>([])

const opts: menubarCheckboxitemsT = [
  { key: "view", trigger: "View", items: ["Sidebar", "Toolbar", "Status Bar"] },
]

<MenubarCheckboxWrapper
  items={opts}
  checked={checked}
  onCheckedChange={(v, c) =>
    setChecked(prev => (c ? [...prev, v] : prev.filter(x => x !== v)))
  }
/>
```

### Radio

```tsx
import { type menubarRadioitemsT, MenubarRadioWrapper } from "@/components/ui/menubar-wrapper"

const [val, setVal] = useState<allowedPrimitiveT>("Light")

const opts: menubarRadioitemsT = [
  { key: "theme", trigger: "Theme", items: ["Light", "Dark", "System"] },
]

<MenubarRadioWrapper items={opts} value={val} onValueChange={setVal} />
```

### Indicator position

```tsx
<MenubarCheckboxWrapper items={opts} indicatorAt="right" />
<MenubarRadioWrapper items={opts} indicatorAt="right" />
```

### Per-menu overrides

Each menu item in the `items` array can override global props:

```tsx
const opts: menubaritemsT = [
  {
    key: "file",
    trigger: "File",
    items: fileItems,
    triggerCls: "font-semibold",
    onSelect: v => handleFileAction(v),
    contentProps: { align: "start" },
  },
  {
    key: "edit",
    trigger: "Edit",
    items: editItems,
  },
]
```

### Complex items (groups, submenus, separators)

```tsx
const opts: menubaritemsT = [
  {
    key: "file",
    trigger: "File",
    items: [
      { label: "New", value: "new", shortcut: "⌘N" },
      "Open",
      "---",
      { group: "Recent", items: ["file1.txt", "file2.txt"] },
      { submenu: "Export", items: ["PDF", "HTML"] },
    ],
  },
]
```

## Reference

#### MenubarWrapper
<AutoTypeTable
name="MenubarWrapper"
type={`export interface MenubarWrapper {
  /**
   * Array of menu configurations — each becomes a separate dropdown in the bar.
   *
   * \`\`\`tsx
   * items={[
   *   { key: "file", trigger: "File", items: ["New", "Open"] },
   *   { key: "edit", trigger: "Edit", items: ["Cut", "Copy"] },
   * ]}
   * \`\`\`
   */
  items: {
    key: string;
    trigger: React.ReactNode;
    items: menuItemsT;
    triggerCls?: string;
    itemCls?: string;
    groupCls?: string;
    groupLabelCls?: string;
    contentProps?: React.ComponentProps<typeof MenubarContent>;
    onSelect?: (value: allowedPrimitiveT) => void;
  }[];

  /**
   * Global trigger CSS class — merged with per-menu triggerCls.
   */
  triggerCls?: string;

  /**
   * Global props forwarded to all MenubarTrigger elements (merged with per-menu triggerProps).
   * Use for render prop (custom element), disabled, etc.
   */
  triggerProps?: Omit<React.ComponentProps<typeof MenubarTrigger>, "children" | "className">;

  /**
   * Global CSS class for all menu items.
   */
  itemCls?: string;

  /**
   * Global CSS class for all groups.
   */
  groupCls?: string;

  /**
   * Global CSS class for all group labels.
   */
  groupLabelCls?: string;

  /**
   * Global props for all MenubarContent components.
   */
  contentProps?: React.ComponentProps<typeof MenubarContent>;

  /**
   * Global callback fired when any item is selected. Per-menu onSelect takes precedence.
   */
  onSelect?: (value: allowedPrimitiveT) => void;

  /**
   * ...rootProps: All Menubar Root props forwarded to the root.
   */
  otherProps?: never;
}`}
/>
