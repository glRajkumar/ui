## DatePicker

### Single (default)

```tsx
import { DatePicker } from "@/components/ui/date-picker"

const [date, setDate] = React.useState<Date | undefined>()

<DatePicker
  selected={date}
  onSelect={setDate}
  startMonth={new Date(2020, 0)}
  endMonth={new Date(2030, 11)}
/>
```

### Range

Closes automatically once both `from` and `to` are selected:

```tsx
import { type DateRange } from "@daypicker/react"

const [range, setRange] = React.useState<DateRange | undefined>()

<DatePicker
  mode="range"
  selected={range}
  onSelect={setRange}
  startMonth={new Date(2020, 0)}
  endMonth={new Date(2030, 11)}
/>
```

### Multiple

Stays open — no auto-close when toggling individual dates:

```tsx
const [dates, setDates] = React.useState<Date[] | undefined>()

<DatePicker
  mode="multiple"
  selected={dates}
  onSelect={setDates}
/>
```

### Custom date format

`dateFormat` is a [date-fns format string](https://date-fns.org/docs/format):

```tsx
<DatePicker selected={date} onSelect={setDate} dateFormat="MMM dd, yyyy" />
```

### Custom trigger

`triggerProps` forwards to the trigger `Button` — accepts all `Button` props except `children`:

```tsx
<DatePicker
  triggerProps={{ variant: "ghost", className: "w-52 border border-dashed" }}
  placeholder="Pick a date..."
  selected={date}
  onSelect={setDate}
/>
```

### Controlled open state

```tsx
const [open, setOpen] = React.useState(false)

<DatePicker open={open} onOpenChange={setOpen} selected={date} onSelect={setDate} />
```

### Popup alignment

```tsx
<DatePicker align="center" selected={date} onSelect={setDate} />
{/* align: "start" | "center" | "end" — default "start" */}
```

### Dropdown navigation bounds

`captionLayout="dropdown"` is the default. Use `startMonth` / `endMonth` to limit the month/year dropdown range:

```tsx
<DatePicker
  startMonth={new Date(2020, 0)}
  endMonth={new Date(2030, 11)}
  selected={date}
  onSelect={setDate}
/>
```

### Label caption

```tsx
<DatePicker captionLayout="label" selected={date} onSelect={setDate} />
```

## DateTimePicker

Combines the calendar with a scrollable time slot panel in the same popover. Single mode only. Changing the day preserves the selected time; changing the time slot preserves the selected day.

### Basic

```tsx
import { DateTimePicker } from "@/components/ui/date-picker"

const [date, setDate] = React.useState<Date | undefined>()

<DateTimePicker
  selected={date}
  onSelect={setDate}
  startMonth={new Date(2020, 0)}
  endMonth={new Date(2030, 11)}
/>
```

### 12-hour format

```tsx
<DateTimePicker
  selected={date}
  onSelect={setDate}
  hourFormat={12}
/>
```

### Minute step

Controls the interval between time slots. Defaults to `5`:

```tsx
<DateTimePicker selected={date} onSelect={setDate} minuteStep={15} />
<DateTimePicker selected={date} onSelect={setDate} minuteStep={30} />
```

Allowed values: `1 | 5 | 10 | 15 | 20 | 30`.

### Custom date format

Defaults to `"dd/MM/yyyy HH:mm"` (24h) or `"dd/MM/yyyy hh:mm a"` (12h):

```tsx
<DateTimePicker selected={date} onSelect={setDate} dateFormat="MMM dd, yyyy HH:mm" />
```

### Disable past

Blocks all past dates in the calendar and all past time slots on today:

```tsx
<DateTimePicker selected={date} onSelect={setDate} disablePast />
```

### Date range constraint

```tsx
<DateTimePicker
  selected={date}
  onSelect={setDate}
  minDate={new Date(2025, 0, 1)}
  maxDate={new Date(2025, 11, 31)}
/>
```

### Minimum datetime

Disables dates before `minTime`'s calendar day, and disables time slots before `minTime` on that day. Compose with `disablePast` — the later constraint wins:

```tsx
const earliest = addMinutes(new Date(), 30)

<DateTimePicker
  selected={date}
  onSelect={setDate}
  minTime={earliest}
  disablePast
/>
```

### Now / Clear / Done

The footer always shows three actions:
- **Now** — snaps selection to the current time (rounded to `minuteStep`)
- **Clear** — resets the value to `undefined`
- **Done** — closes the popover

### Controlled open state

```tsx
<DateTimePicker open={open} onOpenChange={setOpen} selected={date} onSelect={setDate} />
```

## Reference

### DatePicker

<AutoTypeTable
name="DatePicker"
type={`export interface DatePicker {
  /**
   * Selection mode.
   * - "single": picks one date, closes on selection
   * - "range": picks from+to pair, closes when both are set
   * - "multiple": toggles individual dates, never auto-closes
   * @default "single"
   */
  mode?: "single" | "range" | "multiple";

  /**
   * Currently selected value. Type depends on mode:
   * - single:   Date | undefined
   * - range:    DateRange | undefined
   * - multiple: Date[] | undefined
   */
  selected?: Date | DateRange | Date[] | undefined;

  /**
   * Called when the selection changes.
   */
  onSelect?: (value: Date | DateRange | Date[] | undefined) => void;

  /**
   * Controlled open state. Omit to use internal state.
   */
  open?: boolean;

  /**
   * Called when the popover open state changes.
   */
  onOpenChange?: (open: boolean) => void;

  /**
   * Text shown in the trigger when no date is selected.
   * @default "Pick a date"
   */
  placeholder?: string;

  /**
   * date-fns format string for the trigger display.
   * @default "dd/MM/yyyy"
   */
  dateFormat?: string;

  /**
   * Props forwarded to the trigger Button — all Button props except children.
   */
  triggerProps?: Omit<ButtonProps, "children">;

  /**
   * Alignment of the popup relative to the trigger.
   * @default "start"
   */
  align?: "start" | "center" | "end";

  /**
   * ...calendarProps: All Calendar / DayPicker props forwarded —
   * startMonth, endMonth, captionLayout, showWeekNumber, numberOfMonths, etc.
   */
  otherProps?: never;
}`}
/>
