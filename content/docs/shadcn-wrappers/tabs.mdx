## Usage

### Basic

```tsx
import { TabsWrapper } from "@/components/ui/tabs"

<TabsWrapper
  defaultValue="profile"
  tabs={[
    { value: "profile", trigger: "Profile", content: "Manage your profile." },
    { value: "account", trigger: "Account", content: "Update account settings." },
    { value: "security", trigger: "Security", content: "Change your password." },
  ]}
/>
```

### Variants

`default` uses a sliding pill indicator. `line` uses a sliding underline:

```tsx
<TabsWrapper tabs={tabs} defaultValue="profile" variant="default" />
<TabsWrapper tabs={tabs} defaultValue="profile" variant="line" />
```

### Icons in trigger

```tsx
<TabsWrapper
  tabs={[
    { value: "profile",  trigger: <><User /> Profile</>,   content: "..." },
    { value: "settings", trigger: <><Settings /> Settings</>, content: "..." },
  ]}
  defaultValue="profile"
/>
```

### Controlled

```tsx
const [value, setValue] = useState("profile")

<TabsWrapper
  tabs={tabs}
  value={value}
  onValueChange={setValue}
/>
```

### Orientation

```tsx
<TabsWrapper tabs={tabs} defaultValue="profile" orientation="vertical" />
```

Vertical layout places the tab list on the left and content to the right. Arrow key navigation adjusts automatically.

### Activation mode

By default, arrow keys move focus and Enter/Space activates the tab. `activateOnFocus` activates on arrow key press immediately — no Enter needed:

```tsx
<TabsWrapper tabs={tabs} defaultValue="profile" activateOnFocus />
```

### Disabled tab

```tsx
<TabsWrapper
  tabs={[
    { value: "profile",  trigger: "Profile",  content: "..." },
    { value: "account",  trigger: "Account",  content: "...", disabled: true },
    { value: "security", trigger: "Security", content: "..." },
  ]}
  defaultValue="profile"
/>
```

### CSS classes

```tsx
<TabsWrapper
  tabs={tabs}
  defaultValue="profile"
  listCls="w-full"
  triggerCls="flex-1"
  contentCls="mt-4"
/>
```

Per-tab override with `triggerCls` / `contentCls` on each item:

```tsx
<TabsWrapper
  tabs={[
    { value: "a", trigger: "Normal",  content: "...", },
    { value: "b", trigger: "Special", content: "...", triggerCls: "text-primary", contentCls: "bg-muted p-4 rounded" },
  ]}
  defaultValue="a"
/>
```

## Reference

#### tabItemT
<AutoTypeTable
name="tabItemT"
type={`export type tabItemT = {
  /**
   * Unique identifier — used for value matching and as React key.
   */
  value: string;

  /**
   * Content inside the tab button. String or any ReactNode.
   */
  trigger: React.ReactNode;

  /**
   * Content shown when this tab is active.
   */
  content: React.ReactNode;

  /**
   * Prevents the tab from being selected. Focus skips disabled tabs.
   */
  disabled?: boolean;

  /**
   * CSS class for this tab's trigger, overrides the global triggerCls.
   */
  triggerCls?: string;

  /**
   * CSS class for this tab's panel, overrides the global contentCls.
   */
  contentCls?: string;
}`}
/>

#### TabsWrapper
<AutoTypeTable
name="TabsWrapper"
type={`export interface TabsWrapper {
  /**
   * Array of tab items. Each item defines the tab trigger and panel.
   */
  tabs: tabItemsT;

  /**
   * Visual style of the tab list.
   * - default: muted background with sliding pill indicator
   * - line: transparent with sliding underline indicator
   * @default "default"
   */
  variant?: "default" | "line";

  /**
   * CSS class for the tab list container (wraps all triggers).
   */
  listCls?: string;

  /**
   * CSS class applied to every trigger. Per-item triggerCls overrides this.
   */
  triggerCls?: string;

  /**
   * CSS class applied to every panel. Per-item contentCls overrides this.
   */
  contentCls?: string;

  /**
   * Activate the focused tab on arrow key press without needing Enter/Space.
   * @default false
   */
  activateOnFocus?: boolean;

  /**
   * Whether keyboard focus loops from the last tab back to the first.
   * @default true
   */
  loopFocus?: boolean;

  /**
   * ...rootProps: All Base UI Tabs.Root props forwarded to the root.
   * Includes: value, defaultValue, onValueChange, orientation.
   */
  otherProps?: never;
}`}
/>
